def test_password_valid(validator):
    assert validator.validate_password("Aa1!bb") is True


def test_password_not_string_invalid(validator):
    assert validator.validate_password(["Aa1!bb"]) is False
    assert "Parola trebuie sa fie de forma string" in validator.get_password_errors(["Aa1!bb"])


def test_password_with_5_characters_invalid(validator):
    assert validator.validate_password("Aa1!b") is False


def test_password_with_6_characters_valid(validator):
    assert validator.validate_password("Aa1!bb") is True


def test_password_with_20_characters_valid(validator):
    password = "Aa1!" + "b" * 16
    assert len(password) == 20
    assert validator.validate_password(password) is True


def test_password_with_21_characters_invalid(validator):
    password = "Aa1!" + "b" * 17
    assert len(password) == 21
    assert validator.validate_password(password) is False


def test_password_without_lowercase_invalid(validator):
    assert validator.validate_password("AA1!BB") is False


def test_password_without_uppercase_invalid(validator):
    assert validator.validate_password("aa1!bb") is False


def test_password_without_digit_invalid(validator):
    assert validator.validate_password("Aaaa!bb") is False


def test_password_without_special_character_invalid(validator):
    assert validator.validate_password("Aa1234") is False


def test_password_with_space_invalid(validator):
    assert validator.validate_password("Aa1! bb") is False


def test_common_password_invalid(validator):
    assert validator.validate_password("Pass@123") is False


def test_common_password_check_case_insensitive(validator):
    assert validator.is_common_password("PASS@123") is True


@pytest.mark.parametrize("common_password", [
    "123456",
    "admin",
    "12345678",
    "123456789",
    "password",
    "Aa123456",
    "1234567890",
    "Pass@123",
    "admin123",
    "1234567",
    "qwerty123",
    "ParolaCeaMaiTare123!",
    "Parola",
    "Parola123",
])
def test_all_default_common_passwords_are_detected(validator, common_password):
    assert validator.is_common_password(common_password) is True


def test_uppercase_x_is_not_treated_as_special_character(validator):
    assert validator.validate_password("abX123") is False


def test_password_errors_returns_multiple_errors(validator):
    errors = validator.get_password_errors("ana")

    assert "Parola trebuie sa aiba cel putin 6 caractere" in errors
    assert "Parola trebuie sa contina cel putin o litera mare" in errors
    assert "Parola trebuie sa contina cel putin o cifra" in errors
    assert "Parola trebuie sa contina cel putin un caracter special" in errors


# ============================================================
# PASSWORD STRENGTH TESTS
# ============================================================

def test_password_strength_invalid_type_zero(validator):
    assert validator.password_strength_score(None) == 0


def test_password_strength_weak_short_password(validator):
    assert validator.password_strength_score("abc") == 0
    assert validator.password_strength_label("abc") == "slaba"


def test_password_strength_weak_length_only(validator):
    assert validator.password_strength_score("abcdefgh") == 1
    assert validator.password_strength_label("abcdefgh") == "slaba"


def test_password_strength_medium_upper_lower_digit(validator):
    assert validator.password_strength_score("Abcdefgh1") == 3
    assert validator.password_strength_label("Abcdefgh1") == "medie"


def test_password_strength_strong_valid_password(validator):
    assert validator.password_strength_score("Abcdefgh1!") == 4
    assert validator.password_strength_label("Abcdefgh1!") == "puternica"


def test_password_strength_strong_long_password(validator):
    assert validator.password_strength_score("Abcdefgh123!") == 5
    assert validator.password_strength_label("Abcdefgh123!") == "puternica"


def test_common_password_strength_limited_to_one(validator):
    assert validator.password_strength_score("Pass@123") == 1
    assert validator.password_strength_label("Pass@123") == "slaba"

def test_registration_errors_contains_all_invalid_fields(validator):
    errors = validator.get_registration_errors(
        username="ad",
        email="wrong-email",
        password="ana",
        age=15,
        accepted_terms=False,
    )

    assert "username" in errors
    assert "email" in errors
    assert "password" in errors
    assert "age" in errors
    assert "accepted_terms" in errors
